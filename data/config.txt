# Profile: car, bike, or hiker (sets SmartBeacon parameters)
profile=car

# Callsign (APRS)
callsign=ON9RFG

# APRS symbol (2 chars: table + code, e.g. L> = LoRa car)
symbol=L>

# APRS comment appended to position reports
comment=https://RF.Guru

# TX power in dBm (bare modules). Ignored on module-PA boards.
power=23

# Power amplifier present
hasPa=true

# LoRa frequency (MHz)
loraFrequency=433.775

# Radio chip: auto, sx1276 (RFM95) or sx1262
radioChip=auto

# SX1262 clock: auto, 0 (crystal), or TCXO volts e.g. 1.8 / 3.3
loraTcxo=auto

# Radio module: auto, minif27 (internal PA) or bare
radioModule=auto

# PA drive 0-9 on module-PA boards (9 = full output, ~28.7 dBm)
paDrive=9

# GPS baud: 0 = detect, or pin it (9600 u-blox, 115200 ATGM336H)
gpsBaud=0

# --- Voltage Monitoring ---
voltage=true
triggerVoltage=true
triggerVoltageLevel=1200
triggerVoltageCall=ON9RFG
triggerVoltageKeepalive=3600

# --- I2C Sensors ---
i2cEnabled=true
i2cDevice=BME680
bme680TempOffset=0

# --- Debug ---
fullDebug=false
