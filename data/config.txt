# ============================================================
# RF.Guru LoRa 433 APRS Tracker Configuration
# Edit this file and reboot the device to apply changes
# ============================================================

# Profile: car, bike, or hiker (sets SmartBeacon parameters)
profile=car

# Callsign (APRS)
callsign=ON9RFG

# APRS symbol (2 chars: table + code, e.g. L> = LoRa car)
symbol=L>

# APRS comment appended to position reports
comment=https://RF.Guru

# TX power in dBm (bare modules). Ignored on module-PA boards.
power=23

# Power amplifier present
hasPa=true

# LoRa frequency (MHz)
loraFrequency=433.775

# Radio chip: auto, sx1276 (RFM95) or sx1262
radioChip=auto

# SX1262 clock: auto, 0 (crystal), or TCXO volts e.g. 1.8 / 3.3
loraTcxo=auto

# Radio module: auto, minif27 (internal PA) or bare
radioModule=auto

# Chip dBm driving the module PA, -9..22 (22 = rated output)
paDrive=22

# GPS baud: 0 = detect, or pin it (9600 u-blox, 115200 ATGM336H)
gpsBaud=0

# SX1262 internal regulator: dcdc or ldo
loraRegulator=dcdc

# SX1262 PA current limit in mA, max 157
loraOcp=140

# --- Voltage Monitoring ---
voltage=true
triggerVoltage=true
triggerVoltageLevel=1200
triggerVoltageCall=ON9RFG
triggerVoltageKeepalive=3600

# --- I2C Sensors ---
i2cEnabled=true
i2cDevice=BME680
bme680TempOffset=0

# --- USB ---
# Drive used while a computer has the tracker enumerated, -9..8.
# Higher transmits but is not decoded, so the firmware clamps it.
usbPaDrive=-9
# Or refuse to transmit at all while a computer is attached.
usbTxInhibit=false

# --- Battery (thresholds measured on a production vehicle) ---
# Arms itself only after it has seen the alternator, so a tracker on USB
# or a bench supply never mistakes the 12.0V ADC floor for a flat battery.
battProtect=true
battLowVoltage=1200
battChargeVoltage=1300
battPaDrive=10
# 0 for an EV - no starter motor, so no crank dip to ride out
battStartDelay=10
# Cranking is a dip AND a rise; stay off the air while the rail moves.
# Step between 1s samples (V*100) and seconds to wait afterwards. 0 = off.
battStepVolts=30
battSettle=5

# --- Alerts (no console in a car - put the diagnostics on the air) ---
# Both destinations optional; both empty = off. Costs airtime, so it is
# opt-in. alertChannel is a MeshCore public channel whose key is derived
# from its name, e.g. #mbox. alertMention is prefixed to the text.
alertChannel=
alertMention=
alertAprsCall=
# MeshCom destination for the same alerts, as a private message: a callsign,
# a group number, or * for "Messages to All". Needs meshComEnabled=true.
alertMeshComCall=
alertInterval=60

# --- Parked ---
# Stationary this long (seconds) and the beacon rate drops. 0 = off.
sbParkedAfter=14400
sbParkedRate=1800

# --- Airtime ---
# At SF12 a frame is ~3.5s on air. Each of these saves ~10%.
# timestamp off sends '!' instead of '@ddhhmmz'; receivers stamp on
# arrival anyway, and this tracker cannot receive messages.
aprsTimestamp=true
aprsAltitude=true
# Seconds between comments - it is a third of the frame, and aprs.fi
# keeps the last one it saw for 7 days. Or "always" for every beacon,
# or "off" for none at all - telemetry and /A= are unaffected. A station
# that describes itself on MeshCom instead can leave APRS silent here and
# still keep "comment=" set: the MeshCom position picks it up.
commentInterval=1800

# --- MeshCore (IARU R1 ham profile, 70cm) ---
# Mints an Ed25519 identity in /meshid.bin on first enable.
meshEnabled=false
meshName=NOCALL
# chat or repeater are gated to APRS-IS; sensor and room are not
meshNodeType=chat
# direct still reaches APRS-IS without re-flooding the mesh
meshRoute=direct
meshInterval=900

# --- MeshCom (tx-only position beacon and messages) ---
meshComEnabled=false
# seconds, or "smart" to follow the APRS SmartBeacon (60s floor)
meshComInterval=900
meshComMaxHop=2
# messages and HEY carry their own hop count upstream; positions use the
# shorter one above
meshComHopText=4
# the "who hears me" probe, every 15 minutes as a normal node settles on
meshComHey=true
# Station name, sent as #name after the comment. Every real node carries
# one, and a node without a comment still sends "#name".
meshComName=
# Free text, max 25 chars. { } : ; , / are dropped, and a space truncates
# the field at the receiver - that is why stations write Meshcom-Region
# rather than "Meshcom Region". The text is sent verbatim, so it is worth
# saying which network it came out of - the callsign alone does not, e.g.
# meshComComment=vMesh. Empty falls back to "Meshcom".
meshComComment=
# Group subscriptions, e.g. 9;262; - sent as /R= at the end of the position.
# Empty is legitimate: upstream omits the field entirely without groups, and
# this radio cannot receive, so a declared group asks the server to route
# traffic at a station that will never hear it.
meshComGroups=
# 1=TLORA_V2. This board is not in MeshCom's hardware table, so it borrows
# the closest plain SX127x node - 0 means "no info" and may not be shown.
meshComHardwareId=1
# MeshCom's own profile - note the sync word differs from APRS and is
# handled in firmware, not here
meshComFrequency=433.175
meshComBandwidth=250.0
meshComSf=11
meshComCr=6
# 8 = the EU8 country profile the ham networks on this band actually run.
# 32 is MeshCom's "EU" default and makes this the only station on the air
# with a different profile - visible as country 0 in the modulation byte.
meshComPreamble=8

# --- Debug ---
fullDebug=false
